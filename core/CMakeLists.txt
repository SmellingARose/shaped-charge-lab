add_library(scl_core STATIC
  src/ranges.cpp
  src/analysis.cpp
  src/geometry.cpp
  src/devices.cpp
  src/solver.cpp
  src/run.cpp
  src/cpu/cpu_solver.cpp
)
if(NOT APPLE)
  target_sources(scl_core PRIVATE src/gpu/gpu_host.cpp)   # loads the scl_gpu library at runtime
  target_link_libraries(scl_core PRIVATE ${CMAKE_DL_LIBS})
endif()
target_include_directories(scl_core PUBLIC include PRIVATE src)
find_package(Threads REQUIRED)
target_link_libraries(scl_core PUBLIC Threads::Threads)
# CPU optimisation flags apply to C++ only (not to Metal/HIP/CUDA sources)
if(NOT MSVC)
  target_compile_options(scl_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3 -ffast-math -fno-finite-math-only>)
  include(CheckCXXCompilerFlag)
  check_cxx_compiler_flag(-march=native SCL_HAS_MARCH_NATIVE)
  if(SCL_HAS_MARCH_NATIVE AND NOT APPLE)
    target_compile_options(scl_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-march=native>)
  endif()
else()
  target_compile_options(scl_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/O2 /fp:fast /arch:AVX2>)
endif()

add_executable(scl_smoke ${PROJECT_SOURCE_DIR}/tools/smoke.cpp)

add_executable(scl_stephash ${PROJECT_SOURCE_DIR}/tools/stephash.cpp)   # bit-for-bit check of solver changes

# ---- Metal backend (macOS): physics headers + entry points spliced into one MSL source string
if(APPLE)
  option(SCL_WITH_METAL "Build the Metal backend" ON)
endif()
if(SCL_WITH_METAL)
  enable_language(OBJCXX)
  set(_phys ${CMAKE_CURRENT_SOURCE_DIR}/src/physics)
  set(_metal_in ${CMAKE_CURRENT_SOURCE_DIR}/src/metal/kernels.metal.in)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_phys}/portable.h ${_phys}/sph.h ${_metal_in})
  file(READ ${_phys}/portable.h _a)
  file(READ ${_phys}/sph.h _b)
  file(READ ${_metal_in} _c)
  set(_src "${_a}\n${_b}\n${_c}")
  string(REPLACE "#pragma once" "" _src "${_src}")
  string(REPLACE "#include \"portable.h\"" "" _src "${_src}")
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/gen/metal_source.inc "R\"SCLMETAL(${_src})SCLMETAL\"\n")
  target_sources(scl_core PRIVATE src/metal/metal_solver.mm)
  target_include_directories(scl_core PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/gen)
  set_source_files_properties(src/metal/metal_solver.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_compile_definitions(scl_core PUBLIC SCL_WITH_METAL=1)
  target_link_libraries(scl_core PUBLIC "-framework Metal" "-framework Foundation")
endif()

# ---- NVIDIA / AMD backend: one source (src/hip/hip_solver.hip), built as HIP when a HIP toolchain
# is present (AMD ROCm / HIP SDK, or NVIDIA via HIP), otherwise as native CUDA for NVIDIA.
set(SCL_GPU "AUTO" CACHE STRING "NVIDIA/AMD backend: AUTO, HIP, CUDA or OFF")
set_property(CACHE SCL_GPU PROPERTY STRINGS AUTO HIP CUDA OFF)
include(CheckLanguage)
set(_gpu_lang "")
if(SCL_GPU STREQUAL "HIP" OR SCL_GPU STREQUAL "AUTO")
  check_language(HIP)
  if(CMAKE_HIP_COMPILER)
    set(_gpu_lang HIP)
  endif()
endif()
if(NOT _gpu_lang AND (SCL_GPU STREQUAL "CUDA" OR SCL_GPU STREQUAL "AUTO"))
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    set(_gpu_lang CUDA)
  endif()
endif()
if(_gpu_lang)
  enable_language(${_gpu_lang})
  # Its own shared library, loaded by the core at runtime from the executable's folder (src/gpu/gpu_plugin.h)
  set_source_files_properties(src/hip/hip_solver.hip PROPERTIES LANGUAGE ${_gpu_lang})
  add_library(scl_gpu MODULE src/hip/hip_solver.hip)
  set_target_properties(scl_gpu PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})   # next to scl_smoke
  add_dependencies(scl_smoke scl_gpu)
  if(_gpu_lang STREQUAL "CUDA")
    find_package(CUDAToolkit REQUIRED)
    target_link_libraries(scl_gpu PRIVATE CUDA::cudart_static)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
      set_property(TARGET scl_gpu PROPERTY CUDA_ARCHITECTURES native)
    endif()
    set_source_files_properties(src/hip/hip_solver.hip PROPERTIES COMPILE_OPTIONS "-O3;--use_fast_math")
  else()
    set_source_files_properties(src/hip/hip_solver.hip PROPERTIES COMPILE_OPTIONS "-O3;-ffast-math")
  endif()
  message(STATUS "Shaped Charge Lab: GPU backend via ${_gpu_lang}")
else()
  message(STATUS "Shaped Charge Lab: no HIP/CUDA toolchain, GPU backend for NVIDIA/AMD disabled")
endif()
